cmake_minimum_required(VERSION 3.16)
project(ghostshell VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Curses REQUIRED)
add_library(ghostshell_core
  src/config.cpp src/logo.cpp src/metrics.cpp src/processes.cpp
  src/network_storage.cpp)
target_include_directories(ghostshell_core PUBLIC include ${CURSES_INCLUDE_DIRS})
target_compile_options(ghostshell_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(ghostshell src/main.cpp src/ui.cpp)
target_link_libraries(ghostshell PRIVATE ghostshell_core ${CURSES_LIBRARIES})
if(BUILD_TESTING)
  add_executable(ghostshell_tests tests/test_parsing.cpp)
  target_link_libraries(ghostshell_tests PRIVATE ghostshell_core)
  add_test(NAME ghostshell_tests COMMAND ghostshell_tests)
endif()
install(TARGETS ghostshell RUNTIME DESTINATION bin)
